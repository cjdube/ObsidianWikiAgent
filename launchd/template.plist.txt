<!-- Reference template for adding a new vault's scheduled ingest job. Not
     itself loaded by launchd — copy it to a real <label>.plist (concrete
     .plist files are gitignored, so they stay local to your machine), fill in
     the placeholders, then follow the README's "Adding a new vault" steps.

     Placeholders to replace:
       <ABSOLUTE_PATH_TO_REPO>  where you cloned this repo
       <VAULT_NAME>             short name for the vault (e.g. learnings)
       <ABSOLUTE_PATH_TO_VAULT> the vault directory the job is pointed at
       <VAULT_BASENAME>         the LAST PATH COMPONENT of that vault directory

     The last two are usually different, and the difference is why WREN_RUN_LOG
     used to name a file that never existed: VAULT_NAME is a short label you
     pick, while setup_logger names the structured log after the vault
     directory's own basename. install.sh derives both, so run it rather than
     filling these in by hand. -->
<plist version="1.0">
<dict>
    <key>Label</key>
    <string>local.wikiagent.<VAULT_NAME>-ingest</string>

    <key>ProgramArguments</key>
    <array>
        <string><ABSOLUTE_PATH_TO_REPO>/.venv/bin/python</string>
        <string><ABSOLUTE_PATH_TO_REPO>/wiki_ingest.py</string>
        <string>--vault</string>
        <string><ABSOLUTE_PATH_TO_VAULT></string>
    </array>

    <key>WorkingDirectory</key>
    <string><ABSOLUTE_PATH_TO_REPO></string>

    <key>StartCalendarInterval</key>
    <dict>
        <key>Hour</key>
        <integer>9</integer>
        <key>Minute</key>
        <integer>0</integer>
    </dict>

    <key>StandardOutPath</key>
    <string><ABSOLUTE_PATH_TO_REPO>/logs/<VAULT_NAME>-ingest.launchd.log</string>
    <key>StandardErrorPath</key>
    <string><ABSOLUTE_PATH_TO_REPO>/logs/<VAULT_NAME>-ingest.launchd.log</string>

    <!-- WIKI_LAUNCHD_LOG: same path again, so the script can cap the file it is
         writing to. launchd owns that descriptor and nothing rotates it
         otherwise; the learnings job's had reached 20.5 MB. Must match the two
         paths above.

         WREN_RUN_LOG: optional, and only useful if you also run LocalLLMAgent —
         its dashboard can report on this repo's jobs, and reads run history from
         the STRUCTURED log named here. Drop the key if you don't. -->
    <key>EnvironmentVariables</key>
    <dict>
        <key>WIKI_LAUNCHD_LOG</key>
        <string><ABSOLUTE_PATH_TO_REPO>/logs/<VAULT_NAME>-ingest.launchd.log</string>
        <key>WREN_RUN_LOG</key>
        <string><ABSOLUTE_PATH_TO_REPO>/logs/wiki_ingest.<VAULT_BASENAME>.log</string>
    </dict>

    <key>RunAtLoad</key>
    <false/>
</dict>
</plist>
